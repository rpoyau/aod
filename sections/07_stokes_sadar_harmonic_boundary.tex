
\section{Stokes, SADAR and harmonic boundary values}
\label{sec:stokes-sadar-harmonic}
\label{sec:afc-stokes-seed}
\label{subsec:afc-stokes-identity}
The finite flux and response constructions retain their declared channel histories.
For a committed value trace of \eqref{eq:wave-identity}, the harmonic
representation begins at \eqref{eq:fourier-transform}.
\subsection{Typed finite flux}
\paragraph{Setup.}
AFC declares normalized memory \(p_i^{\mathrm{mem}}\ge0\), \(\sum_i p_i^{\mathrm{mem}}=1\),
and a stochastic kernel \(P\). A\(\Omega\)D separately resolves pressure-bearing
values of admitted returned channels. Keep their types explicit:
\begin{equation}
\phi^{\mathrm{AFC}}_{ij}=p_i^{\mathrm{mem}}P_{ij}-p_j^{\mathrm{mem}}P_{ji},
\qquad \phi^{\mathrm{AOD}}_{ij}(B)=p^D_{ij}(B)A_{ij}(B).
\label{eq:flux-types}
\end{equation}
Each normalized product lies in \([0,1]\), hence
\(|\phi^{\mathrm{AFC}}_{ij}|\le1\). Pressure values such as \(7\) are A\(\Omega\)D
channel values in their declared count domain. An oriented ledger channel
has a reverse view of equal magnitude and opposite sign. A separately
constructed reverse physical channel retains its own directional RCD; it
is not identified with that bookkeeping reverse view.

\paragraph{Joint invariance.}
For a relabelling \(g\), \(p^{\mathrm{mem}}_{gi}=p^{\mathrm{mem}}_i\) and
\(P_{gi,gj}=P_{ij}\) imply \(\phi^{\mathrm{AFC}}_{gi,gj}=\phi^{\mathrm{AFC}}_{ij}\)
by substitution. Kernel symmetry alone supplies no such result: a two-state
swap kernel with memory \((3/4,1/4)\) has flux \(1/2\), whereas equal memory
gives zero. In both cases an admitted closure retains its ancestry and one phase.

\paragraph{Region and boundary.}
Let \(V\) be the finite vertex set, \(R\subseteq V\), and
\(\partial R=\{(i,j):i\in R,j\notin R\}\) its oriented cut.
\begin{equation}
\operatorname{div}(i;B)=\sum_{j\in V}\phi_{ij}(B).
\label{eq:b-scoped-divergence}
\end{equation}
\paragraph{Finite Stokes: derivation.}
For either one declared antisymmetric flux type, split the double sum:
\begin{equation}
\begin{split}
\sum_{i\in R}\operatorname{div}(i;B)
&=\sum_{i,j\in R}\phi_{ij}(B)+\sum_{i\in R,j\notin R}\phi_{ij}(B)\\
&=\sum_{i\in R,j\notin R}\phi_{ij}(B).
\end{split}
\label{eq:afc-stokes-used}
\end{equation}
The internal sum is zero because each ordered pair cancels its reverse and
diagonal entries are zero. Cancellation removes a summed value, not the
two contributing lineage records. The same finite argument holds componentwise.
\begin{equation}
\phi^{\mathrm{AOD}}_{ij,\AO}(B)=p^D_{ij,\AO}(B)A_{ij,\AO}(B).
\label{eq:afc-flux-ao-resolution}
\end{equation}
In the remaining pressure formulas \(\phi\) abbreviates \(\phi^{\mathrm{AOD}}\).
\paragraph{Falsification.} Wrong antisymmetry or an omitted boundary channel
breaks the finite identity; it does not create an alternative admission rule.
\paragraph{Provenance.} AFC Choices (memory, stochastic kernel, flux) and
Stokes corollary, finite internal-pair cancellation (AFC \texttt{choice:memory-state}, \texttt{choice:kernel}, \texttt{choice:flux}, \texttt{choice:divergence}, \texttt{choice:region}, \texttt{cor:afc-stokes}, \texttt{eq:afc-stokes-cor}); A\(\Omega\)D
\eqref{eq:rcd-coupling} and \eqref{eq:b-scoped-channel-attention}.
\subsubsection{\texorpdfstring{\(B\)-scoped flux contents}{B-scoped flux contents}}
\label{subsec:b-scoped-flux-contents}
Use the pressure-resolved A\(\Omega\)D flux notation on the declared boundary/window
\[
B=(\partial F,\omega).
\]
The scoped flux is written \(\phi_{ij}(B)\). The declared calculation fixes the contents of \(\phi_{ij}(B)\).

For a declared path-history family \(\mathcal H_{ij}(B)\), the path-history pressure is
\begin{equation}
p^D_{ij}(B)
=
\sum_{h\in\mathcal H_{ij}(B)}
C_{3,h}\rho^D_{\omega,h}(B).
\label{eq:path-history-pressure}
\end{equation}
A compressed channel form is licensed only after the declared compression
\begin{equation}
\mathsf{Comp}_{ij}:
\{(C_{3,h},RD_h,R^{\mathrm{asym}}_h,A_h)\}_{h\in\mathcal H_{ij}(B)}
\mapsto
(C_{3,ij},\rho^D_{ij},A_{ij}).
\label{eq:path-history-compression}
\end{equation}
Under that declared compression,
\begin{equation}
p^D_{ij}(B)=C_{3,ij}\rho^D_{\omega,ij}(B)
=
C_{3,ij}\min\!\left(\rho^D_{ij}(B),|\omega|_{ij}\right).
\label{eq:b-scoped-channel-pressure}
\end{equation}
The bare-duration form is the neutral-reflection reduction:
\[
R^{\mathrm{asym}}_{ij}(B)=\mathrm{id}
\quad\Rightarrow\quad
\rho^D_{ij}(B)=D_{ij}(B).
\]
The direction term is
\begin{equation}
A_{ij}(B)=\Delta\chi^{\mathrm{lag}}_{ij}(B)i_{ij}(B).
\label{eq:b-scoped-channel-attention}
\end{equation}
Then
\begin{equation}
\phi_{ij}(B)=p^D_{ij}(B)A_{ij}(B).
\label{eq:b-scoped-flux}
\end{equation}
Equivalently,
\begin{equation}
\phi_{ij}(B)=
C_{3,ij}\rho^D_{\omega,ij}(B)
\Delta\chi^{\mathrm{lag}}_{ij}(B)i_{ij}(B).
\label{eq:b-scoped-flux-expanded}
\end{equation}
\paragraph{Magnitude and direction.}
The scoped flux factors as
\[
\phi_{ij}(B)=\underbrace{p^D_{ij}(B)}_{\mathrm{magnitude}}
\underbrace{A_{ij}(B)}_{\mathrm{direction}}.
\]
If \(p^D_{ij}(B)\mapsto\lambda p^D_{ij}(B)\) while \(A_{ij}(B)\) is unchanged, then \(\phi_{ij}(B)\mapsto\lambda\phi_{ij}(B)\). For the same declared channel family, \(\mathbf A_F(B)\mapsto\lambda\mathbf A_F(B)\).

The oriented ledger channel \((i,j)\) fixes \(\phi_{ij}(B)\); its reverse bookkeeping view is fixed by antisymmetry:
\begin{equation}
\phi_{ji}(B)=-\phi_{ij}(B),
\qquad
\phi_{ii}(B)=0.
\label{eq:b-scoped-flux-antisymmetry}
\end{equation}
When \(i_{ij}(B)\) has components, the antisymmetry and Stokes identity are evaluated componentwise.

The divergence is the sum defined in \eqref{eq:b-scoped-divergence}.

For \(R\subseteq V\), Finite Stokes for this antisymmetric A\(\Omega\)D ledger on the declared scope \(B\) is
\begin{equation}
\sum_{i\in R}\operatorname{div}(i;B)
=
\sum_{\substack{i\in R\\j\notin R}}\phi_{ij}(B).
\label{eq:b-scoped-stokes}
\end{equation}

\subsubsection{Slice flux}
\label{subsec:slice-flux}
\begin{definition}[Duon slice flux]
For a declared cut-frame \(t\), let
\[
B|_t=(\partial F,\{t\}).
\]
The duon-current slice flux is the one cut-frame crossing
\begin{equation}
\Phi^D_{\partial R}(t;B|_t)
=
\sum_{e\in\partial R(t)}
\phi_e(B|_t).
\label{eq:slice-flux}
\end{equation}
When the scope is fixed, write \(\Phi^D_{\partial R}(t)\). For raw duon-current content, \(\phi_e(B|_t)=\epsilon_e C_{3,e}\).
\end{definition}

\subsubsection{Window current}
\label{subsec:window-current}
\begin{definition}[Window duon current]
The window current is the retained many-frame continuation
\begin{equation}
J^D_{\partial R}(\omega;B)
=
\sum_{t\in\omega}
\Phi^D_{\partial R}(t;B|_t).
\label{eq:window-current}
\end{equation}
When the scope is fixed, write \(J^D_{\partial R}(\omega)\).
\end{definition}
A slice flux is one declared cut-frame crossing of duon/trit current through the boundary. A window current is the retained many-frame continuation of such crossings across the declared window.

The raw capacity crossing count is
\begin{equation}
N^D_{\partial R}(\omega;B)
=
\sum_{t\in\omega}
\sum_{e\in\partial R(t)}
C_{3,e}.
\label{eq:raw-capacity-crossing-count}
\end{equation}
The magnitude current is
\begin{equation}
J^D_{\mathrm{mag}}(\omega;B)
=
\sum_{t\in\omega}
\sum_{e\in\partial R(t)}
p^D_e(B|_t).
\label{eq:magnitude-current}
\end{equation}
When the declared boundary/window partitions current status, require
\begin{equation}
J^D_{\mathrm{mag}}
=
J^D_{\mathrm{in}}+J^D_{\mathrm{out}}+J^D_{\mathrm{pending}}.
\label{eq:current-magnitude-split}
\end{equation}

\subsubsection{Moving boundary}
\label{subsec:moving-boundary}
For a moving retained boundary, the window may record crossing, sweep, and shedding terms as
\begin{equation}
\Phi^D_{\partial R,\mathrm{mov}}(t)
=
\Phi^D_{\mathrm{cross}}(t)
-
\Phi^D_{\mathrm{sweep}}(t)
+
\Phi^D_{\mathrm{shedding}}(t).
\label{eq:moving-boundary}
\end{equation}
Here crossing is duon current crossing the retained boundary, sweep is support swept by boundary motion, and shedding is disturbance that breaks from the retained boundary.

\subsubsection{Regional boundary template}
\label{subsec:regional-boundary-template}
The same boundary/current form applies across familiar region choices. Declare a region, declare \(\partial R\), and account for current crossing, returning, and remaining pending.

\subsubsection{Scope}
\label{subsec:field-count-fixed-scope}
Fix a boundary/window scope
\[
\mathcal B=(\partial F,\omega).
\]
All quantities in this subsection are evaluated with respect to \(\mathcal B\).

\subsubsection{Retained field family}
\label{subsec:retained-field-family}
Let
\[
\mathcal F_{\mathcal B}=\{F_1,\ldots,F_{n_F}\}
\]
be the retained field family, and define
\begin{equation}
n_F(\mathcal B):=|\mathcal F_{\mathcal B}|.
\label{eq:field-count}
\end{equation}
Here \(n_F\) is retained field count, \(T\) is path/account duration or tick count, \(m\) is enclosure order in \(\mathsf C_{2m}\), and \(p\) is rotor or cycle completion period.

\subsubsection{Returned-current channels}
\label{subsec:returned-current-channels}
For \(F_i,F_j\in\mathcal F_{\mathcal B}\), write
\[
\ADAR_{i\leftarrow j}
\]
for the returned-current channel from \(F_j\) into \(F_i\). The diagonal terms are self-return terms; off-diagonal terms are cross-return terms. A retained field family may carry field-ring returned-current relations when \(E_{\mathrm{return}}\) is declared on the fixed boundary/window.

\subsubsection{Field-count term}
\label{subsec:field-count-term}
As a declared count-weighting model, the field-count contribution is
\begin{equation}
\mathcal O_{\partial F}(\mathcal B)=n_F(\mathcal B)\log n_F(\mathcal B),
\label{eq:field-count-term}
\end{equation}
with the logarithm base fixed by the declared calculation. A single scoped organization of an admitted recurrent monon-mode wave has \(n_F=1\). Alias names are resolved before counting; an isolated monon is not a field. Coupled-field counting begins at \(n_F=2\).

\subsubsection{Disoma declaration}
\label{subsec:disoma-declaration}
For a two-field declared scope, fix
\[
B_{12}=(\partial F_{12},\omega_{12}),
\qquad
\mathcal F_{B_{12}}=\{F_1,F_2\}.
\]
Thus
\[
n_F(B_{12})=2,
\qquad
\mathcal O_{\partial F}(B_{12})=2\log 2.
\]

\paragraph{Calculation split.}
Field-count scaling and shell/enclosure routing are connected by the declared boundary/window, but they are separate calculations:
\[
B\to \mathcal F_{B}\to n_F(B)\to \ADAR_{i\leftarrow j}\to \mathcal O_{\partial F}(B)\to \SADAR.
\]
Shell/enclosure routing fixes
\[
\begin{aligned}
\mathcal C_{2m}
&\to X_{\mathrm{shedding}}
\to \mathrm{SheddicPath}(X_{\mathrm{shedding}}),\\
\mathrm{SheddicPath}
&\to
\{\text{rebalancing},\ \text{exoshedding},\ \text{endoshedding},\ \text{redirected support routing}\}.
\end{aligned}
\]

\subsection{Duonic pressure}
\label{sec:duonic-pressure-inertia}

\subsubsection{Scalar burden}
\label{subsec:duonic-pressure-scalar-burden}
\begin{definition}[Channel pressure]
For a returned duon-current channel \(e\) on \(B=(\partial F,\omega)\), channel pressure uses the window-clipped duration participation:
\[
p^D_e(B)=C_{3,e}\rho^D_{\omega,e}(B).
\]
Here \(\rho^D_\omega\) is the window-clipped duration participation. It is distinct from \(TTL_{\mathcal C}\), the seat-compatible retention duration of a support enclosure or sheddic path.
\end{definition}
\begin{definition}[Duonic pressure]
The scalar pressure of a field is
\begin{equation}
P^D_F(B)=
\sum_{e\in F(B)}p^D_e(B).
\label{eq:duonic-pressure}
\end{equation}
It records the retained/clipped duon/trit burden carried on the declared boundary/window; pending overhang is recorded separately by \(Q^D\).
\end{definition}

\subsubsection{Pressure weighting}
\label{subsec:duonic-pressure-weighting}
When \(P^D_F(B)>0\), a normalized orientation \(\theta_F^{(q)}\) may be weighted by scalar pressure to provide scalar factors later used in \(\mathbf A_F(B)\) in \secref{sec:field:sadar}.

\subsubsection{Inertial burden}
\label{subsec:inertial-burden}
Duonic pressure resists change of attention orientation. In the declared inverse-pressure response model with \(P^D_F(B)>0\), a committed perturbing current supplies the next-transition orientation increment:
\[
\Delta\theta_{F,k+1}^{(q)}=
\frac{\Delta J_{\mathrm{duon}}^{(q)}}{P^D_F(B)}.
\]

\subsection{SADAR value operator}
\label{subsec:sadar-operator}
\begin{definition}[SADAR operator]
Write \(\SADARop_B\) for Symmetry Attention Duration Asymmetric Reflection (SADAR) on the declared boundary/window \(B\). For a retained cycle \(C\),
\begin{equation}
\SADARop_B(C)
=
\phi_{C,\AO}(B)
=
p^D_{C,\AO}(B)A_{C,\AO}(B).
\label{eq:sadar-operator-cycle}
\end{equation}
Expanded,
\begin{equation}
\SADARop_B(C)
=
C_{3,C}\rho^D_{\omega,C}(B)\,\Delta\chi_C^{\mathrm{lag}}(B)i_C(B).
\label{eq:sadar-operator-expanded}
\end{equation}
For two committed SADAR-coupled recurrent lineages \(C_u,C_v\),
\begin{equation}
\SADARop_B(C_u,C_v)
=
\phi^{\mathrm{beat}}_{uv,\AO}(B)
=
p^D_{uv,\AO}(B)A_{uv,\AO}(B).
\label{eq:sadar-operator-beat-coupling}
\end{equation}
\end{definition}
If \(\bar e\) is the reversed oriented boundary element of \(e\), then
\begin{equation}
\SADARop_B(C_{\bar e})=-\SADARop_B(C_e).
\label{eq:sadar-orientation-antisymmetry}
\end{equation}


\subsection{SADAR}
\label{sec:field:sadar}
\label{subsec:sadar}

\subsubsection{ADAR channel}
\label{subsec:adar-channel}
\begin{definition}[ADAR]
ADAR is one local returned-current channel:
\begin{equation}
\ADAR_{ij}(B)=(A_{ij}(B),D_{ij}(B),R^{\mathrm{asym}}_{ij}(B)).
\label{eq:adar-channel}
\end{equation}
Here \(A_{ij}(B)\) is returned-current attention orientation, \(D_{ij}(B)\) is returned-current duration, and \(R_{ij}^{\mathrm{asym}}(B)\) is asymmetric reflection on the declared boundary/window. The attention term is \(A_{ij}(B)=\Delta\chi^{\mathrm{lag}}_{ij}(B)i_{ij}(B)\).
\end{definition}

\subsubsection{SADAR balance}
\label{subsec:sadar-balance}
\begin{definition}[SADAR]
SADAR is Symmetric Attention--Duration--Asymmetric Reflection. It is the field's symmetry-chasing returned-current dynamic on a declared boundary/window. For \(B=(\partial F,\omega)\),
\begin{equation}
\SADAR_F(B)=
\big(\ADAR_{ij}(B)\big)_{(i,j)\in F(B)}.
\label{eq:sadar-field}
\end{equation}
\end{definition}
The indexed family denotes the declared returned-current channels on \(B\), taken with their boundary incidence, phase relation, and pressure contribution. The declared field balance is the relative SADAR balance of returned duon current on \(B\). The field chases this balance through attention, duration, asymmetric reflection, phase bias, reseating, and shedding.

\subsubsection{SADAR attention vector}
\label{subsec:sadar-attention-vector}
\begin{definition}[SADAR attention vector]
The SADAR attention vector is the boundary/window-specific codification of SADAR dynamics. It records pressure-weighted response input of returned duon current on the declared boundary/window:
\begin{equation}
\mathbf A_F(B)=
\sum_{(i,j)\in F(B)}
\phi_{ij}(B).
\label{eq:sadar-attention-vector}
\end{equation}
Equivalently,
\begin{equation}
\mathbf A_F(B)=
\sum_{(i,j)\in F(B)}
C_{3,ij}\rho^D_{\omega,ij}(B)
\Delta\chi^{\mathrm{lag}}_{ij}(B)i_{ij}(B).
\label{eq:sadar-attention-vector-expanded}
\end{equation}
\end{definition}
The scoped A\(\Omega\)D flux \(\phi_{ij}(B)\) carries SADAR contents. The product \(C_{3,ij}\rho^D_{\omega,ij}(B)\) is the duonic-pressure contribution on \(B\); \(\Delta\chi^{\mathrm{lag}}_{ij}(B)i_{ij}(B)\) is the phase-bias / boundary-orientation contribution.

\subsubsection{Phase bias, push/pull, and motion}
\label{subsec:phase-bias-motion}
Phase bias is the signed lead--lag imbalance of returned-current channels. Push and pull are signed phase-biased components of the same SADAR attention vector.
\paragraph{Realized motion.}
The pressure-bearing attention vector is an input to the declared update.
Motion is its realized state transition:

\begin{equation}
\operatorname{motion}_{F,k+1}(B)=(S^W_{F,k},S^W_{F,k+1}),\qquad
S^W_{F,k+1}=\mathcal U^{\SADAR}_B(S^W_{\le k},S^\times_{\le k},\mathbf A_{F,\le k}).
\label{eq:causal-update}
\end{equation}
All pressure, lag and directional-duration inputs are committed through
\(k\); the response is at \(k+1\). No coordinate is generated by the same
response that consumes it. For a declared rational example with \(p_k>0\),
\begin{equation}
J_{k+1}=J_k+\mathbf A_k/p_k.
\label{eq:sadar-rational-response}
\end{equation}
This realization leaves identity and closure memory fixed unless a separate
admissible closure transition is witnessed. A zero scalar residual alone is
not such a witness. The support triples AO and OA exchange branches; their
oriented signed values obey \(AO=-OA\), with the hinge count retained.\label{eq:ao-oa-oriented-value}

\paragraph{Statement: incident return shift.}
If \(\mathbf A_F(B)=0\) and an unmatched incident returned duon \(e\) contributes
\[
\phi_e(B)=p^D_e(B)A_e(B),
\]
then
\begin{equation}
\mathbf A_{F+e}(B)=\phi_e(B).
\label{eq:incident-return-shift}
\end{equation}
If a matching return \(\bar e\) is declared and \(\phi_{\bar e}(B)=-\phi_e(B)\), then
\begin{equation}
\mathbf A_{F+e+\bar e}(B)=0.
\label{eq:matched-return-cancellation}
\end{equation}

\paragraph{Statement: short window and high incident current.}
If \(|\omega|_h<\rho^D_h(B)\), then \(\rho^D_{\omega,h}(B)=|\omega|_h\) and \(Q^D_h(B)>0\). For a short window,
\[
p^D_h(B)=C_{3,h}|\omega|_h.
\]
Many incident histories or large \(C_{3,h}\) can still shift the SADAR attention vector when
\[
\sum_h p^D_h(B)A_h(B)\ne0.
\]

The channel-level rate form is
\[
\Delta\phi_{ij}=p^D_{ij}\Delta A_{ij}
\]
when retained pressure is fixed across the compared restrictions. Pending overhang separates from clipped pressure:
\begin{equation}
Q^{D,\mathrm{dur}}_{ij}(B)=\max\!\left(0,\rho^D_{ij}(B)-|\omega|_{ij}\right),
\label{eq:duration-overhang}
\end{equation}
\begin{equation}
Q^D_{ij}(B)=C_{3,ij}Q^{D,\mathrm{dur}}_{ij}(B).
\label{eq:burden-overhang}
\end{equation}
A falling field may remain internally balanced relative to its own boundary/window while its SADAR attention vector is phase-biased relative to another declared boundary.

\subsubsection{Self-return and cross-return}
\label{subsec:self-cross-return}
A single declared field carries self-return symmetry:
\[
\ADAR_{i\leftarrow i}=\text{self-return channel}.
\]
When another field is included in the same accounting scope, cross-return channels are added:
\[
\ADAR_{i\leftarrow j}=\text{cross-return channel},\qquad i\ne j.
\]
Self-return and cross-return are channel forms inside the same SADAR balance. A field ring gate is a finite cross-return fixture with declared boundary gates, returned-current relations, and output bins.

\subsubsection{Boundary-relative simultaneity}
\label{subsec:boundary-relative-simultaneity}
Boundary-relative simultaneity names the coexistence of component-field and enclosing-field SADAR balances. Component fields and enclosing fields may each carry their own SADAR attention vector because each vector is scoped by its declared boundary/window. For the enclosing two-field scope, write
\[
\mathcal{B}_{12}=(\partial F_{12},\omega_{12}).

\subsection{Finite harmonic coordinates of admitted recurrence}

\paragraph{Setup.}
Let \(G\) be a declared finite abelian index group of order \(N>0\), and
\(\widehat G\) its exact character group. Each \(\chi:G\to\mathbb C\)
satisfies \(\chi(g+h)=\chi(g)\chi(h)\) and \(|\chi(g)|=1\).
Fix an admitted recurrent lineage and a value trace \(v:G\to\mathbb C\)
bound to its committed history. Translation means \(T_a v(g)=v(g-a)\).
Using \(G\) to index a finite record is a representation choice; periodic
extension of that record supplies no proof of periodic dynamics.

\paragraph{Definition and normalization.}
\begin{equation}
\widehat v(\chi)=\frac1N\sum_{g\in G}v(g)\overline{\chi(g)},
\qquad \langle v,w\rangle_G=\frac1N\sum_{g\in G}v(g)\overline{w(g)}.
\label{eq:fourier-transform}
\end{equation}
Exact characters are downstream coordinates of the admitted trace. Neither
the transform nor its inversion is a closure, motif or wave admission test.

\paragraph{Orthogonality: derivation.}
For a nontrivial character \(\eta\), choose \(a\) with \(\eta(a)\ne1\).
Translation of the finite sum gives \(S=\sum_g\eta(g)=\eta(a)S\), hence
\(S=0\). For the trivial character the sum is \(N\). Thus
\begin{equation}
\frac1N\sum_{g\in G}\chi(g)\overline{\xi(g)}
=\mathbf1_{\chi=\xi},\qquad
\frac1N\sum_{\chi\in\widehat G}\chi(g)\overline{\chi(h)}
=\mathbf1_{g=h}.
\label{eq:fourier-orthogonality}
\end{equation}
For a cyclic factor of order \(L\), its \(L\) characters are the powers of
an exact primitive \(L\)-th root; products over a cyclic decomposition give
exactly \(|\widehat G|=N\) characters. Orthogonality therefore gives a basis
of the \(N\)-dimensional function space and the second identity by the same
translation argument on the dual. This establishes completeness.

\paragraph{Inversion and uniqueness: derivation.}
Substitute \eqref{eq:fourier-transform} and apply the second identity:
\begin{equation}
\sum_{\chi}\widehat v(\chi)\chi(g)
=\frac1N\sum_h v(h)\sum_\chi\chi(g)\overline{\chi(h)}
=v(g).
\label{eq:fourier-inversion}
\end{equation}
Taking the inner product with any character recovers its coefficient, so
no second coefficient family represents a different trace with the same
full transform. Distinct provenance can still accompany equal value traces.
\paragraph{Falsification.} A missing coefficient, altered normalization or
unbound trace fails the corresponding representation or admission premise.
\paragraph{Provenance.} Finite character sum and
\eqref{eq:fourier-orthogonality}; admission remains
\eqref{eq:wave-identity} and \eqref{eq:motif-admission}.

\paragraph{Linearity: setup and derivation.}

\widehat{av+bw}=a\widehat v+b\widehat w.
\label{eq:fourier-linearity}
\end{equation}
\paragraph{Specialization.} A declared sum of admitted value traces is

\paragraph{Falsification.} Applying this value identity to infer a linear

\label{eq:fourier-convolution-product}
\end{equation}

\label{eq:fourier-correlation-power}
\end{equation}
For \(v=w\), the power coefficients are \(N|\widehat v(\chi)|^2\).
\paragraph{Falsification.} Equal power is insufficient for equal complex
coefficients, equal phase, or equal construction provenance.
\paragraph{Provenance.} Equation \eqref{eq:fourier-orthogonality}.

\paragraph{Parseval: derivation.}
At zero shift the same calculation, or direct orthogonality, gives
\begin{equation}
\frac1N\sum_g v(g)\overline{w(g)}
=\sum_\chi\widehat v(\chi)\overline{\widehat w(\chi)},
\qquad \frac1N\sum_g|v(g)|^2=\sum_\chi|\widehat v(\chi)|^2.
\label{eq:fourier-parseval}
\end{equation}
\paragraph{Specialization.} This is a squared-value identity in the chosen
coordinate space; a physical energy meaning requires its own declared map.
\paragraph{Falsification.} Changing normalization on one side breaks equality.
\paragraph{Provenance.} Equation \eqref{eq:fourier-orthogonality}.

\paragraph{Real traces and standing coordinates: derivation.}
For real \(v\), complex conjugation of the defining sum gives
\begin{equation}
\widehat v(\overline\chi)=\overline{\widehat v(\chi)}.
\label{eq:fourier-real-symmetry}
\end{equation}
For a non-self-conjugate pair define real coordinates
\begin{equation}
c_\chi=(\chi+\overline\chi)/2,\quad
s_\chi=(\chi-\overline\chi)/(2\mathrm i),\qquad
a\chi+\overline a\,\overline\chi
=2\operatorname{Re}(a)c_\chi-2\operatorname{Im}(a)s_\chi.
\label{eq:fourier-running-standing}
\end{equation}
Running character and standing pair coordinates represent the same admitted
trace. Self-conjugate characters are already real and occur once.
\paragraph{Falsification.} This coordinate change cannot classify an open
word as a running wave or establish physical propagation.
\paragraph{Provenance.} Equations \eqref{eq:fourier-inversion} and
\eqref{eq:fourier-real-symmetry}.

\paragraph{Recurrence subgroup: setup and derivation.}
Let \(H\le G\), and assume the bound value trace is invariant under every
translation in \(H\). Shift covariance gives
\((1-\overline{\chi(h)})\widehat v(\chi)=0\). Therefore
\begin{equation}
v(g+h)=v(g)\ (h\in H)
\quad\Longleftrightarrow\quad
\operatorname{supp}\widehat v\subseteq H^\perp,
\quad H^\perp=\{\chi:\chi(h)=1\ \forall h\in H\}.
\label{eq:fourier-recurrence-support}
\end{equation}
The converse follows by translating each term in the inversion.
\paragraph{Specialization.} Exact periodic values on a declared cyclic
index may satisfy this hypothesis. Jittered cadence need not do so.
\paragraph{Falsification.} A finite observation window alone supplies no
translation invariance or exact dynamical recurrence subgroup.
\paragraph{Provenance.} Equations \eqref{eq:fourier-shift-modulation} and
\eqref{eq:fourier-inversion}.

\paragraph{Direct products: setup and derivation.}
For \(G=G_1\times\cdots\times G_s\), characters factor as
\(\chi(g)=\prod_j\chi_j(g_j)\), and \(N=\prod_j|G_j|\).
If, additionally, the trace is separable, finite sums factor:
\begin{equation}
v(g)=\prod_j v_j(g_j)
\quad\Longrightarrow\quad
\widehat v(\chi_1,\ldots,\chi_s)=\prod_j\widehat v_j(\chi_j).
\label{eq:fourier-product-factorization}
\end{equation}
\paragraph{Specialization.} Declared independent recurrence orders may use
a product group. A coupled trace generally has nonseparable coefficients.
\paragraph{Falsification.} Product index structure alone does not imply a
product trace. Linearity and finite exact reconstruction do not eliminate
nonlinear coupling or establish motif admission.
\paragraph{Provenance.} Equation \eqref{eq:fourier-transform}, finite products
of cyclic character bases, and the declared separability premise.
\subsubsection{Finite cancellation residuals}
\label{subsec:afc-flux-stokes-audit}
For a declared region \(R\) and scope \(B\), define
\begin{equation}
S_{\mathrm{int}}(R;B)
=
\sum_{i\in R}\sum_{j\in R}\phi_{ij}(B),
\label{eq:stokes-audit-internal}
\end{equation}
\begin{equation}
S_{\mathrm{div}}(R;B)
=
\sum_{i\in R}\operatorname{div}(i;B),
\qquad
S_{\partial}(R;B)
=
\sum_{\substack{i\in R\\j\notin R}}\phi_{ij}(B).
\label{eq:stokes-audit-sums}
\end{equation}
A valid declaration satisfies internal cancellation,
\begin{equation}
S_{\mathrm{int}}(R;B)=0,
\label{eq:stokes-audit-internal-zero}
\end{equation}
and boundary balance,
\begin{equation}
\mathcal E(R,B)
=
S_{\mathrm{div}}(R;B)-S_{\partial}(R;B)=0.
\label{eq:stokes-audit-residual}
\end{equation}
If \(S_{\mathrm{int}}(R;B)\ne0\), the declared orientation, sign, or antisymmetry fails on that scope. If \(\mathcal E(R,B)\ne0\), the declaration has a missing returned-current channel, incomplete boundary/window, undeclared sheddic path, or wrong channel content.
